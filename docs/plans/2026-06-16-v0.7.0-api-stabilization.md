> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 引入分组 Functional Options 模式，让 `NewAgent` 在构造时一次性注入所有能力，构造后核心不可变，消除 14 个 Deprecated 字段的直接使用。

**Tech Stack:** Go 1.22+, 标准库 only, TDD (Red-Green-Refactor)

```go
// agentprimordia/internal/agent/config.go
package agent

import (
	"fmt"
	"log/slog"

	"agentprimordia/internal/llm"
	"agentprimordia/internal/memory"
	"agentprimordia/internal/persist"
	"agentprimordia/internal/tools"
)

// AgentConfig 是 Agent 的完整配置，构造后不可变。
// 用户不直接构造此 struct，而是通过 Option 函数填充。
type AgentConfig struct {
	// ===== 核心必填 =====
	Name          string
	SystemPrompt  string
	Model         llm.Provider
	PromptTemplate *PromptTemplate

	// ===== 标量配置 =====
	MaxTurns    int
	Temperature float64
	SessionID   string

	// ===== 分组能力配置 =====
	Memory        MemoryConfig
	RAG           RAGConfig
	Observability ObservabilityConfig
	Resilience    ResilienceConfig
	Tools         ToolsConfig

	// ===== 运行时辅助 =====
	Lifecycle *Lifecycle
	Logger    *slog.Logger
}

// MemoryConfig 记忆能力配置
type MemoryConfig struct {
	Store      MemoryStore
	Summarizer memory.SummaryExtractor
	FileScope  []string
}

// ObservabilityConfig 可观测性配置
type ObservabilityConfig struct {
	Hooks       Hooks
	Tracer      Tracer
	Metrics     MetricsRecorder
	Events      EventPublisher
	CostTracker *CostTracker
}

// ResilienceConfig 容错配置
type ResilienceConfig struct {
	CheckpointStore persist.CheckpointStore
	HITL            *HITLConfig
	Cache           llm.LLMCache
	ContextWindow   ContextWindowStrategy
}

// ToolsConfig 工具配置
type ToolsConfig struct {
	Registry *tools.Registry
}

// defaultConfig 返回带默认值的 AgentConfig
func defaultConfig() AgentConfig {
	return AgentConfig{
		MaxTurns:    50,
		Temperature: 0.0,
		Lifecycle:   NewLifecycle(),
		Logger:      slog.Default(),
	}
}

// Validate 集中校验配置
func (c *AgentConfig) Validate() error {
	if c.Name == "" {
		return fmt.Errorf("agent name is required")
	}
	if c.Model == nil {
		return fmt.Errorf("agent model (LLM Provider) is required")
	}
	if c.MaxTurns <= 0 {
		return fmt.Errorf("MaxTurns must be positive, got %d", c.MaxTurns)
	}
	return nil
}
```

```go
// agentprimordia/internal/agent/options.go
package agent

import (
	"agentprimordia/internal/llm"
	"agentprimordia/internal/memory"
	"agentprimordia/internal/persist"
	"agentprimordia/internal/tools"
)

// Option 是 Agent 构造选项
type Option func(*AgentConfig)

// ===== 标量配置 =====

// WithMaxTurns 设置最大循环轮数（默认 50）
func WithMaxTurns(n int) Option {
	return func(c *AgentConfig) { c.MaxTurns = n }
}

// WithTemperature 设置 LLM 采样温度（默认 0.0）
func WithTemperature(t float64) Option {
	return func(c *AgentConfig) { c.Temperature = t }
}

// WithSessionID 设置会话 ID
func WithSessionID(id string) Option {
	return func(c *AgentConfig) { c.SessionID = id }
}

// WithPromptTemplate 设置自定义提示词模板
func WithPromptTemplate(t *PromptTemplate) Option {
	return func(c *AgentConfig) { c.PromptTemplate = t }
}

// ===== 顶层快捷注入（对应原 Deprecated 字段）=====

// WithMemory 注入记忆存储
func WithMemory(m MemoryStore) Option {
	return func(c *AgentConfig) { c.Memory.Store = m }
}

// WithToolkit 注入工具注册表
func WithToolkit(r *tools.Registry) Option {
	return func(c *AgentConfig) { c.Tools.Registry = r }

func WithHooks(h Hooks) Option {

// WithRAG 注入检索增强配置
func WithRAG(cfg RAGConfig) Option {
```

```go
// 替换 agentprimordia/internal/agent/new_agent.go 的内容
package agent

import (
	"fmt"

	"agentprimordia/internal/llm"
)

// NewAgent 创建 Agent，构造后核心能力不可变。
// 这是从 v0.7.0 起的推荐入口。
//
// 必填参数：
//   - name: Agent 名称
//   - systemPrompt: 系统提示词（可为空）
//   - model: LLM Provider
//
// 可选能力通过 Option 函数注入，例如：
//
//	agent, err := agent.NewAgent("assistant", "你是助手", llm,
//	    agent.WithMemory(memStore),
//	    agent.WithToolkit(registry),
//	    agent.WithMaxTurns(15),
//	)
func NewAgent(name, systemPrompt string, model llm.Provider, opts ...Option) (*CapabilityAgent, error) {
	cfg := defaultConfig()
	cfg.Name = name
	cfg.SystemPrompt = systemPrompt
	cfg.Model = model
	for _, opt := range opts {
		opt(&cfg)
	}
	if err := cfg.Validate(); err != nil {
		return nil, fmt.Errorf("invalid agent config: %w", err)
	}
	return buildAgent(cfg)
}

// buildAgent 从配置构建 Agent（内部函数）
func buildAgent(cfg AgentConfig) (*CapabilityAgent, error) {
	// 构造 ReActAgent（复用现有逻辑）
	reactCfg := ReActConfig{
		Name:           cfg.Name,
		SystemPrompt:   cfg.SystemPrompt,
		PromptTemplate: cfg.PromptTemplate,
		Model:          cfg.Model,
		MaxTurns:       cfg.MaxTurns,
		Temperature:    cfg.Temperature,
		SessionID:      cfg.SessionID,
		Lifecycle:      cfg.Lifecycle,
		Logger:         cfg.Logger,
	}
	a := NewReActAgent(reactCfg)

	// 注入能力（通过现有链式 API，保持双写兼容）
	cap := a.AsCapability()

	if cfg.Memory.Store != nil {
		cap = cap.WithMemory(cfg.Memory.Store)
	}
	if cfg.Memory.Summarizer != nil {
		cap = cap.WithSummarizer(cfg.Memory.Summarizer)
	}
	if len(cfg.Memory.FileScope) > 0 {
		cap = cap.WithFileScope(cfg.Memory.FileScope)
	}
	if cfg.Tools.Registry != nil {
		cap = cap.WithToolkit(cfg.Tools.Registry)
	}
	if cfg.Observability.Hooks != nil {
		cap = cap.WithHooks(cfg.Observability.Hooks)
	}
	if cfg.Observability.Tracer != nil {
		cap = cap.WithTracer(cfg.Observability.Tracer)
	}
	if cfg.Observability.Metrics != nil {
		cap = cap.WithMetrics(cfg.Observability.Metrics)
	}
	if cfg.Observability.Events != nil {
		cap = cap.WithEvents(cfg.Observability.Events)
	}
	if cfg.Observability.CostTracker != nil {
		cap = cap.WithCostTracker(cfg.Observability.CostTracker)
	}
	if cfg.Resilience.CheckpointStore != nil {
		cap = cap.WithCheckpointStore(cfg.Resilience.CheckpointStore)
	}
	if cfg.Resilience.HITL != nil {
		cap = cap.WithHITL(*cfg.Resilience.HITL)
	}
	if cfg.Resilience.Cache != nil {
		cap = cap.WithCache(cfg.Resilience.Cache)
	}
	if cfg.Resilience.ContextWindow != nil {
		cap = cap.WithContextWindow(cfg.Resilience.ContextWindow)
	}
	if cfg.RAG.Enabled {
		cap = cap.WithRAG(cfg.RAG)
	}

	return cap, nil
}
```
